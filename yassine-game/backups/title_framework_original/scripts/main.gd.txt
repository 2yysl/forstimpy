extends Node2D

const GAME_SCENE := "res://scenes/game.tscn"
const MINIMUM_SECONDS := 3.0
const PAN_SECONDS := 1.2

@onready var camera: Camera2D = $Camera2D
@onready var game_view: Marker2D = $GameView
@onready var game_host: Node2D = $GameHost
@onready var message: RichTextLabel = $Message


func _ready() -> void:
	var started_at := Time.get_ticks_msec()
	var screen_size := get_viewport_rect().size

	# Start one screen above the game's starting view.
	camera.global_position = (
		game_view.global_position
		- Vector2(0.0, screen_size.y)
	)
	camera.make_current()

	# Center the loading sentence in that upper area.
	message.size = Vector2(screen_size.x, 120.0)
	message.global_position = (
		camera.global_position
		- Vector2(screen_size.x / 2.0, 60.0)
	)

	message.bbcode_enabled = true
	message.scroll_active = false
	message.add_theme_font_size_override("normal_font_size", 36)

	message.text = (
        "[center]"
		+ LoadingMessages.choose_message()
		+ " [wave amp=4 freq=3]↻[/wave]"
		+ "[/center]"
	)

	# Hold normal gameplay processing until the pan finishes.
	game_host.process_mode = Node.PROCESS_MODE_DISABLED

	# Allow the loading screen to draw.
	await get_tree().process_frame

	# Begin loading the game in the background.
	var error := ResourceLoader.load_threaded_request(
		GAME_SCENE,
        "PackedScene"
	)

	if error != OK:
		show_error("Couldn't start loading the game.")
		return

	# Keep the screen responsive while loading.
	while true:
		var status := ResourceLoader.load_threaded_get_status(
			GAME_SCENE
		)

		if status == ResourceLoader.THREAD_LOAD_LOADED:
			break

		if status != ResourceLoader.THREAD_LOAD_IN_PROGRESS:
			show_error("Couldn't load the game.")
			return

		await get_tree().process_frame

	var packed_scene := ResourceLoader.load_threaded_get(
		GAME_SCENE
	) as PackedScene

	if packed_scene == null:
		show_error("The loaded resource isn't a game scene.")
		return

	# Add the game at its original coordinates.
	var game := packed_scene.instantiate()
	game_host.add_child(game)

	# Keep the message visible for at least three seconds total.
	var elapsed := (
		Time.get_ticks_msec() - started_at
	) / 1000.0

	var remaining := MINIMUM_SECONDS - elapsed

	if remaining > 0.0:
		await get_tree().create_timer(remaining).timeout

	# Pan down: text exits upward, game enters from below.
	var tween := create_tween()
	tween.set_trans(Tween.TRANS_CUBIC)
	tween.set_ease(Tween.EASE_IN_OUT)

	tween.tween_property(
		camera,
		"global_position",
		game_view.global_position,
		PAN_SECONDS
	)

	await tween.finished

	message.hide()
	game_host.process_mode = Node.PROCESS_MODE_INHERIT

	# Optional: tell your game's root script the intro is finished.
	if game.has_method("start_game"):
		game.call("start_game")


func show_error(text: String) -> void:
	message.text = "[center]" + text + "[/center]"
	push_error(text)
